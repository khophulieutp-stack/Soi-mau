function doGet(e) { return ContentService.createTextOutput("Hệ thống nhận dữ liệu giám sát nguồn sáng đang hoạt động!"); } function doPost(e) { try { // 1. Lấy dữ liệu từ Web App gửi lên var data = JSON.parse(e.postData.contents); var light = data.light; // "TL83" hoặc "D65" var date = data.date; var start = data.start; var end = data.end; var minutes = Number(data.minutes) || 0; var user = data.user || ""; var note = data.note || ""; // 2. Mở Sheet tương ứng var ss = SpreadsheetApp.getActiveSpreadsheet(); var sheet = ss.getSheetByName(light); if (!sheet) { return ContentService.createTextOutput("Lỗi: Không tìm thấy Sheet " + light).setMimeType(ContentService.MimeType.TEXT); } // Lấy giới hạn tuổi thọ var maxLife = (light === "TL83") ? 120000 : 12000; // 3. TÌM CHÍNH XÁC DÒNG TIÊU ĐỀ "STT" var columnA = sheet.getRange("A:A").getValues(); var headerRow = 0; var lastActualRow = 0; // Quét từ trên xuống để tìm dòng có chứa chữ "STT" for (var i = 0; i < columnA.length; i++) { var val = String(columnA[i][0]).trim().toUpperCase(); if (val === "STT") { headerRow = i + 1; // Vì index mảng bắt đầu từ 0, số dòng bắt đầu từ 1 break; } } // Nếu không tìm thấy chữ STT, báo lỗi if (headerRow === 0) { return ContentService.createTextOutput("Lỗi: Không tìm thấy chữ STT ở cột A trong Sheet " + light).setMimeType(ContentService.MimeType.TEXT); } // Tìm dòng cuối cùng có dữ liệu số (dưới dòng STT) for (var j = columnA.length - 1; j >= headerRow; j--) { if (columnA[j][0] !== "" && !isNaN(columnA[j][0])) { lastActualRow = j + 1; break; } } var stt = 1; var prevTotal = 0; // Nếu bảng đã có dữ liệu trước đó if (lastActualRow > 0) { stt = Number(sheet.getRange(lastActualRow, 1).getValue()) + 1; prevTotal = Number(sheet.getRange(lastActualRow, 7).getValue()) || 0; } // Nếu bảng chưa có dữ liệu (chỉ mới có tiêu đề STT) else { lastActualRow = headerRow; } // 4. Tính toán số liệu mới var newTotal = prevTotal + minutes; var newRemain = maxLife - newTotal; // 5. Ghi dữ liệu vào DÒNG TIẾP THEO (ngay dưới dòng cuối cùng) var targetRow = lastActualRow + 1; var rowData = [ [stt, date, light, start, end, minutes, newTotal, newRemain, user, note] ]; // Sử dụng setValues để ghi chính xác vào 10 cột của dòng mục tiêu sheet.getRange(targetRow, 1, 1, 10).setValues(rowData); return ContentService.createTextOutput("Success").setMimeType(ContentService.MimeType.TEXT); } catch (error) { return ContentService.createTextOutput("Lỗi: " + error.toString()).setMimeType(ContentService.MimeType.TEXT); } }
